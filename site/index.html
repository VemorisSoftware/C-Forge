factorial.cfv
// Funciones y recursión
funcion factorial(n) {
    si (n <= 1) {
        retornar 1
    }
    retornar n * factorial(n - 1)
}

// Closures capturan el entorno
funcion crear_sumador(base) {
    retornar funcion(valor) {
        retornar base + valor
    }
}

sea sumar_diez = crear_sumador(10)

afirmar(factorial(6) == 720, "factorial(6)")
afirmar(sumar_diez(32) == 42, "closure ok")
mostrar(factorial(6))   // → 720

variables.cfv
// Declarar variables con 'sea'
sea nombre = "C-Forge"
sea version = 3.7

// Bucle con acumulador
sea suma = 0
sea i    = 1

mientras (i <= 10) {
    suma = suma + i
    i    = i + 1
}

mostrar(suma)          // → 55
afirmar(suma == 55, "suma 1..10")
mostrar("[OK] bucles")
funciones.cfv
funcion factorial(n) {
    si (n <= 1) { retornar 1 }
    retornar n * factorial(n - 1)
}

funcion potencia(base, exp) {
    sea resultado = 1
    sea i = 0
    mientras (i < exp) {
        resultado = resultado * base
        i = i + 1
    }
    retornar resultado
}

mostrar(factorial(6))     // → 720
mostrar(potencia(2, 8)) // → 256
closures.cfv
funcion crear_sumador(base) {
    retornar funcion(valor) {
        retornar base + valor
    }
}

funcion crear_multiplicador(factor) {
    retornar funcion(x) {
        retornar x * factor
    }
}

sea sumar_diez = crear_sumador(10)
sea doble      = crear_multiplicador(2)

mostrar(sumar_diez(32)) // → 42
mostrar(doble(21))      // → 42
control.cfv
funcion clasificar(n) {
    si (n < 0)  { retornar "negativo" }
    si (n == 0) { retornar "cero" }
    retornar "positivo"
}

mostrar(clasificar(-5))  // → negativo
mostrar(clasificar(0))   // → cero
mostrar(clasificar(42))  // → positivo

// FizzBuzz en C-Forge
sea n = 1
mientras (n <= 15) {
    si (n % 15 == 0) { mostrar("FizzBuzz") }
    si (n % 3  == 0) { mostrar("Fizz") }
    si (n % 5  == 0) { mostrar("Buzz") }
    n = n + 1
}

english.cfv
// In English
function factorial(n) {
    if (n <= 1) {
        return 1
    }
    return n * factorial(n - 1)
}

let result = factorial(6)
print(result)        // → 720
assert(result == 720, "ok")
